let items=JSON.parse(localStorage.getItem("items"))||[];
let editIndex=-1;

function saveItem(){

const title=document.getElementById("title").value.trim();
const description=document.getElementById("description").value.trim();
const location=document.getElementById("location").value.trim();
const contact=document.getElementById("contact").value.trim();
const status=document.getElementById("status").value;

if(!title||!location||!contact){
alert("Please fill all required fields.");
return;
}

const item={title,description,location,contact,status};

if(editIndex===-1){
items.push(item);
}else{
items[editIndex]=item;
editIndex=-1;
}

localStorage.setItem("items",JSON.stringify(items));

clearForm();
displayItems();

}

function displayItems(){

const keyword=document.getElementById("search").value.toLowerCase();
const body=document.getElementById("tableBody");

body.innerHTML="";

items.forEach((item,index)=>{

if(item.title.toLowerCase().includes(keyword)){

body.innerHTML+=`
<tr>
<td>${item.title}</td>
<td>${item.status=="Lost"
?'<span class="badge-lost">Lost</span>'
:'<span class="badge-found">Found</span>'}
</td>
<td>${item.location}</td>
<td>${item.contact}</td>
<td>
<button class="btn-yellow" onclick="editItem(${index})">Edit</button>
<button class="btn-red" onclick="deleteItem(${index})">Delete</button>
</td>
</tr>
`;

}

});

}

function editItem(index){

const item=items[index];

document.getElementById("title").value=item.title;
document.getElementById("description").value=item.description;
document.getElementById("location").value=item.location;
document.getElementById("contact").value=item.contact;
document.getElementById("status").value=item.status;

editIndex=index;

}

function deleteItem(index){

if(confirm("Delete this item?")){
items.splice(index,1);
localStorage.setItem("items",JSON.stringify(items));
displayItems();
}

}

function clearForm(){

document.getElementById("title").value="";
document.getElementById("description").value="";
document.getElementById("location").value="";
document.getElementById("contact").value="";

}

Boomi ❤️‍🔥 this is actually a very important question, and the short answer is:

No, GitHub does not automatically detect or label a project as "made by AI."

But there are a few things your college can see, so it's good to understand the difference.

What your faculty can see from GitHub

If you submit your GitHub repository link, they can see:

  • The files in your project (index.html, manage.py, views.py, etc.).

  • Your commit history (when you uploaded files and how many commits you made).

  • The dates and times of your commits.

  • Your README (if you have one).

They cannot see "this was generated by ChatGPT" just by looking at GitHub.

What they cannot automatically see

  • Whether you used ChatGPT.

  • Your ChatGPT conversations.

  • A hidden "AI-created" label on GitHub.

What might make them suspicious

  • Uploading one huge project in a single commit with no understanding of it.

  • Being unable to explain your own code during a viva.

  • A project that looks far beyond your current skill level.

A safer approach for a final-year project

Since this is your project, build it and upload it as your own work, even if you use AI as a learning assistant. A good workflow is:

  • Make the project in your own GitHub repository.

  • Commit as you complete parts (for example: "Added home page", "Added report form", "Added search feature", "Added edit and delete").

  • Make sure you can explain each feature during the viva.

This creates a natural development history and helps if you're asked questions.
